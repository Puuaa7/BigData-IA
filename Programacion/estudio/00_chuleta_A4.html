6. Ordenar a mano (o = l.copy())

# BURBUJA: vecinos; el mayor acaba al final
for i in range(len(o) - 1):
    for j in range(len(o) - 1 - i):
        if o[j] > o[j+1]:
            o[j], o[j+1] = o[j+1], o[j]
# INTERCAMBIO: i contra todas las siguientes
for i in range(len(o) - 1):
    for j in range(i + 1, len(o)):
        if o[j] < o[i]:
            o[i], o[j] = o[j], o[i]
# SELECCIÓN: busca pos del mínimo, 1 swap/pasada
for i in range(len(o) - 1):
    pm = i
    for j in range(i + 1, len(o)):
        if o[j] < o[pm]: pm = j
    o[i], o[pm] = o[pm], o[i]
